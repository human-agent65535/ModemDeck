cmake_minimum_required(VERSION 3.20)
project(modemdeck_audio_core LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
get_filename_component(MD_REPOSITORY "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(MD_VENDOR "${MD_REPOSITORY}/third_party/audio")
file(READ "${MD_VENDOR}/upstream.json" MD_UPSTREAM_JSON)
string(JSON MD_ARCHIVE_NAME GET "${MD_UPSTREAM_JSON}" archive)
string(JSON MD_ARCHIVE_SHA GET "${MD_UPSTREAM_JSON}" archive_sha256)
file(SHA256 "${MD_VENDOR}/${MD_ARCHIVE_NAME}" MD_ACTUAL_SHA)
if(NOT MD_ACTUAL_SHA STREQUAL MD_ARCHIVE_SHA)
  message(FATAL_ERROR "Vendored audio source checksum mismatch")
endif()
# The Go toolchain incorporates this into CGO_CPPFLAGS, so C-only changes
# invalidate Go's cache even when the external static library path is unchanged.
file(GLOB_RECURSE MD_BUILD_INPUTS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*")
list(SORT MD_BUILD_INPUTS)
set(MD_BUILD_HASH_INPUT "${MD_UPSTREAM_JSON}")
foreach(input IN LISTS MD_BUILD_INPUTS)
  file(SHA256 "${input}" input_sha)
  file(RELATIVE_PATH input_name "${CMAKE_CURRENT_SOURCE_DIR}" "${input}")
  string(APPEND MD_BUILD_HASH_INPUT "${input_name}:${input_sha}\n")
endforeach()
string(SHA256 MD_BUILD_SHA "${MD_BUILD_HASH_INPUT}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/audio-source.sha256" "${MD_BUILD_SHA}\n")
set(MD_WEBRTC_SOURCE "${CMAKE_CURRENT_BINARY_DIR}/upstream/src")
set(MD_UNPACKED_SHA "")
if(EXISTS "${CMAKE_CURRENT_BINARY_DIR}/upstream.sha256")
  file(READ "${CMAKE_CURRENT_BINARY_DIR}/upstream.sha256" MD_UNPACKED_SHA)
endif()
if(NOT MD_UNPACKED_SHA STREQUAL MD_ARCHIVE_SHA)
  file(REMOVE_RECURSE "${CMAKE_CURRENT_BINARY_DIR}/upstream")
  file(ARCHIVE_EXTRACT INPUT "${MD_VENDOR}/${MD_ARCHIVE_NAME}" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/upstream")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/upstream.sha256" "${MD_ARCHIVE_SHA}")
endif()
set(ABSL_PROPAGATE_CXX_STD ON CACHE BOOL "" FORCE)
set(ABSL_BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory("${MD_WEBRTC_SOURCE}/third_party/abseil-cpp" abseil EXCLUDE_FROM_ALL)
set(OPUS_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_PKG_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_CMAKE_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
set(OPUS_ENABLE_FLOAT_API ON CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPUS_MAY_HAVE_NEON OFF CACHE BOOL "" FORCE)
set(OPUS_PRESUME_NEON OFF CACHE BOOL "" FORCE)
set(OPUS_DISABLE_INTRINSICS ON CACHE BOOL "" FORCE)
set(OPUS_DNN OFF CACHE BOOL "" FORCE)
foreach(feature SSE SSE2 SSE4_1 AVX2)
  set(OPUS_X86_MAY_HAVE_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${MD_WEBRTC_SOURCE}/third_party/opus/src" opus EXCLUDE_FROM_ALL)
include(sources.cmake)
add_library(md_neteq STATIC md_neteq.cc ${MD_NETEQ_SOURCES}
  "${MD_WEBRTC_SOURCE}/api/task_queue/default_task_queue_factory_stdlib.cc"
  "${MD_WEBRTC_SOURCE}/rtc_base/task_queue_stdlib.cc")
target_include_directories(md_neteq PUBLIC include PRIVATE "${MD_WEBRTC_SOURCE}"
  "${MD_WEBRTC_SOURCE}/third_party/abseil-cpp" "${MD_WEBRTC_SOURCE}/third_party/opus/src/include")
target_compile_definitions(md_neteq PRIVATE WEBRTC_POSIX WEBRTC_OPUS_SUPPORT_120MS_PTIME
  WEBRTC_OPUS_VARIABLE_COMPLEXITY=0 WEBRTC_STRICT_FIELD_TRIALS=0 NDEBUG)
if(APPLE)
  target_compile_definitions(md_neteq PRIVATE WEBRTC_MAC)
  target_link_libraries(md_neteq PUBLIC "-framework CoreFoundation")
  if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    target_compile_definitions(md_neteq PRIVATE WEBRTC_IOS)
  endif()
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_compile_definitions(md_neteq PRIVATE WEBRTC_LINUX)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
    target_compile_definitions(md_neteq PRIVATE WEBRTC_ARCH_ARM64)
  elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^arm")
    target_compile_definitions(md_neteq PRIVATE WEBRTC_ARCH_ARM)
  endif()
endif()
target_compile_options(md_neteq PRIVATE -Wno-nullability-completeness -Wno-deprecated-declarations
  -ffunction-sections -fdata-sections)
target_link_libraries(md_neteq PUBLIC opus absl::strings absl::str_format absl::any_invocable absl::optional)
if(EMSCRIPTEN)
  target_compile_options(md_neteq PRIVATE -fexceptions)
  # Export the reviewed C header, never C++/WebRTC implementation symbols.
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/include/md_neteq.h" MD_ABI_HEADER)
  string(REGEX MATCHALL "md_[a-z0-9_]+[ \t]*\\(" MD_ABI_MATCHES "${MD_ABI_HEADER}")
  set(MD_EXPORTS "'_malloc','_free'")
  foreach(match IN LISTS MD_ABI_MATCHES)
    string(REGEX REPLACE "[ \t]*\\(" "" symbol "${match}")
    string(APPEND MD_EXPORTS ",'_${symbol}'")
  endforeach()
  add_executable(md_audio_module module.cc)
  target_link_libraries(md_audio_module PRIVATE md_neteq)
  target_link_options(md_audio_module PRIVATE --no-entry -fexceptions -sDISABLE_EXCEPTION_CATCHING=0
    -sALLOW_MEMORY_GROWTH=0 -sINITIAL_MEMORY=33554432 -sSTACK_SIZE=1048576
    -sFILESYSTEM=0 -sMODULARIZE=1 -sEXPORT_NAME=createAudioCore -sEXPORT_ES6=1
    -sWASM_ASYNC_COMPILATION=0 -sENVIRONMENT=web,worker,shell
    "-sEXPORTED_FUNCTIONS=[${MD_EXPORTS}]"
    -sEXPORTED_RUNTIME_METHODS=HEAP16,HEAPF32,HEAPU8,HEAPU32)
  set_target_properties(md_audio_module PROPERTIES OUTPUT_NAME audioCore SUFFIX ".mjs")
else()
  add_custom_target(md_audio_core_bundle ALL
    COMMAND "${CMAKE_COMMAND}" "-DMD_BUILD_DIR=${CMAKE_CURRENT_BINARY_DIR}"
      "-DMD_AR=${CMAKE_AR}" "-DMD_APPLE=${APPLE}" -P "${CMAKE_CURRENT_SOURCE_DIR}/merge-static.cmake"
    DEPENDS md_neteq VERBATIM)
endif()
